import tkinter as tk from tkinter import messagebox # ----------------------------- # QUIZ QUESTIONS # ----------------------------- questions = [ { "question": "Which language is used to create web pages?", "options": ["Python", "HTML", "Java", "C++"], "answer": "HTML" }, { "question": "Which symbol is used to write a comment in Python?", "options": ["//", "#", "/* */", ""], "answer": "#" }, { "question": "Which keyword is used to define a function in Python?", "options": ["function", "define", "def", "fun"], "answer": "def" }, { "question": "Which data type stores True or False?", "options": ["String", "Integer", "Boolean", "Float"], "answer": "Boolean" }, { "question": "Which function is used to display output in Python?", "options": ["display()", "print()", "show()", "output()"], "answer": "print()" }, { "question": "Which of these is a Python collection?", "options": ["List", "Loop", "Function", "Operator"], "answer": "List" }, { "question": "What is the extension of a Python file?", "options": [".java", ".html", ".py", ".cpp"], "answer": ".py" }, { "question": "Which keyword is used to create a loop?", "options": ["repeat", "for", "loop", "iterate"], "answer": "for" }, { "question": "Which operator is used for multiplication in Python?", "options": ["x", "*", "%", "#"], "answer": "*" }, { "question": "Which function is used to get input from the user?", "options": ["get()", "input()", "read()", "scan()"], "answer": "input()" } ] # ----------------------------- # VARIABLES # ----------------------------- current_question = 0 score = 0 student_name = "" # ----------------------------- # MAIN WINDOW # ----------------------------- root = tk.Tk() root.title("Online Quiz Application") root.geometry("700x500") root.configure(bg="#0f172a") root.resizable(False, False) # ----------------------------- # TITLE # ----------------------------- title = tk.Label( root, text="ONLINE QUIZ APPLICATION", font=("Arial", 24, "bold"), bg="#0f172a", fg="white" ) title.pack(pady=25) # ----------------------------- # NAME SECTION # ----------------------------- name_label = tk.Label( root, text="Enter Your Name", font=("Arial", 14), bg="#0f172a", fg="white" ) name_label.pack(pady=5) name_entry = tk.Entry( root, font=("Arial", 14), width=30 ) name_entry.pack(pady=10) # ----------------------------- # START BUTTON # ----------------------------- def start_quiz(): global student_name student_name = name_entry.get().strip() if student_name == "": messagebox.showwarning( "Warning", "Please enter your name!" ) return name_entry.pack_forget() name_label.pack_forget() start_button.pack_forget() show_question() start_button = tk.Button( root, text="START QUIZ", font=("Arial", 14, "bold"), bg="#2563eb", fg="white", width=15, command=start_quiz ) start_button.pack(pady=20) # ----------------------------- # QUESTION LABEL # ----------------------------- question_label = tk.Label( root, text="", font=("Arial", 18, "bold"), bg="#0f172a", fg="white", wraplength=600 ) # ----------------------------- # OPTION VARIABLE # ----------------------------- selected_answer = tk.StringVar() # ----------------------------- # RADIO BUTTONS # ----------------------------- radio_buttons = [] for i in range(4): rb = tk.Radiobutton( root, text="", variable=selected_answer, value="", font=("Arial", 14), bg="#0f172a", fg="white", selectcolor="#1e293b", activebackground="#0f172a", activeforeground="white", anchor="w", width=40 ) radio_buttons.append(rb) # ----------------------------- # QUESTION COUNTER # ----------------------------- question_counter = tk.Label( root, text="", font=("Arial", 12), bg="#0f172a", fg="#94a3b8" ) # ----------------------------- # SHOW QUESTION # ----------------------------- def show_question(): question_label.pack(pady=25) question_counter.pack(pady=5) for rb in radio_buttons: rb.pack(pady=5) next_button.pack(pady=20) selected_answer.set("") question = questions[current_question] question_label.config( text=question["question"] ) question_counter.config( text=f"Question {current_question + 1} of {len(questions)}" ) for i in range(4): radio_buttons[i].config( text=question["options"][i], value=question["options"][i] ) # ----------------------------- # NEXT QUESTION # ----------------------------- def next_question(): global current_question global score answer = selected_answer.get() if answer == "": messagebox.showwarning( "Warning", "Please select an answer!" ) return correct_answer = questions[current_question]["answer"] if answer == correct_answer: score += 1 current_question += 1 if current_question < len(questions): show_question() else: show_result() # ----------------------------- # NEXT BUTTON # ----------------------------- next_button = tk.Button( root, text="NEXT", font=("Arial", 14, "bold"), bg="#16a34a", fg="white", width=15, command=next_question ) # ----------------------------- # SHOW RESULT # ----------------------------- def show_result(): question_label.pack_forget() question_counter.pack_forget() for rb in radio_buttons: rb.pack_forget() next_button.pack_forget() percentage = (score / len(questions)) * 100 result_title = tk.Label( root, text="QUIZ COMPLETED!", font=("Arial", 25, "bold"), bg="#0f172a", fg="#22c55e" ) result_title.pack(pady=40) result_label = tk.Label( root, text=f"Congratulations, {student_name}!\n\n" f"Your Score: {score}/{len(questions)}\n" f"Percentage: {percentage:.2f}%", font=("Arial", 18), bg="#0f172a", fg="white" ) result_label.pack(pady=10) restart_button = tk.Button( root, text="RESTART QUIZ", font=("Arial", 14, "bold"), bg="#2563eb", fg="white", width=15, command=lambda: restart_quiz( result_title, result_label, restart_button ) ) restart_button.pack(pady=30) # ----------------------------- # RESTART QUIZ # ----------------------------- def restart_quiz( result_title, result_label, restart_button ): global current_question global score global student_name current_question = 0 score = 0 student_name = "" result_title.destroy() result_label.destroy() restart_button.destroy() name_label.pack(pady=5) name_entry.delete(0, tk.END) name_entry.pack(pady=10) start_button.pack(pady=20) # ----------------------------- # START APPLICATION # ----------------------------- root.mainloop()
